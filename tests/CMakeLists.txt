function(add_openv6_test name)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE openv6)
    target_compile_definitions(${name} PRIVATE OPENV6_TEST_DATA="${CMAKE_CURRENT_SOURCE_DIR}/data" OPENV6_NATIVE_DIR="${PROJECT_SOURCE_DIR}/native")
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES SKIP_RETURN_CODE 77)
endfunction()
